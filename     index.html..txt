<script>
(() => {
  "use strict";

  const $ = id => document.getElementById(id);

  let state = { tasks: [], points: 0, quizDone: false };
  try {
    const saved = JSON.parse(localStorage.getItem("smartstudy-v2"));
    if (saved && Array.isArray(saved.tasks)) state = {...state, ...saved};
  } catch (_) {}

  function save() {
    try { localStorage.setItem("smartstudy-v2", JSON.stringify(state)); }
    catch (_) {}
    updateStats();
  }

  function updateStats() {
    const level = Math.floor(state.points / 50) + 1;
    $("points").textContent = state.points;
    $("taskCount").textContent = state.tasks.filter(t => !t.done).length;
    $("level").textContent = level;
    $("progressPoints").textContent = state.points;
    $("progressLevel").textContent = level;
    $("progressBar").style.width = (state.points % 50 / 50 * 100) + "%";
  }

  function showPage(id) {
    document.querySelectorAll(".page").forEach(p => p.classList.toggle("active", p.id === id));
    document.querySelectorAll(".nav button").forEach(b => b.classList.toggle("active", b.dataset.page === id));
    window.scrollTo({top: 0, behavior: "smooth"});
  }

  document.querySelectorAll("[data-page]").forEach(b =>
    b.addEventListener("click", () => showPage(b.dataset.page))
  );
  document.querySelectorAll("[data-open]").forEach(b =>
    b.addEventListener("click", () => showPage(b.dataset.open))
  );

  // Домашние задания
  function renderTasks() {
    const list = $("taskList");
    list.replaceChildren();

    state.tasks.forEach((task, index) => {
      const row = document.createElement("div");
      row.className = "task" + (task.done ? " done" : "");

      const check = document.createElement("input");
      check.type = "checkbox";
      check.checked = task.done;
      check.setAttribute("aria-label", "Отметить выполнение");
      check.addEventListener("change", () => {
        if (check.checked && !task.done) state.points += 5;
        if (!check.checked && task.done) state.points = Math.max(0, state.points - 5);
        task.done = check.checked;
        save();
        renderTasks();
      });

      const text = document.createElement("span");
      text.textContent = task.text;

      const remove = document.createElement("button");
      remove.className = "btn secondary";
      remove.type = "button";
      remove.textContent = "Удалить";
      remove.addEventListener("click", () => {
        if (task.done) state.points = Math.max(0, state.points - 5);
        state.tasks.splice(index, 1);
        save();
        renderTasks();
      });

      row.append(check, text, remove);
      list.append(row);
    });
    updateStats();
  }

  $("taskForm").addEventListener("submit", event => {
    event.preventDefault();
    const value = $("taskInput").value.trim();
    if (!value) return;
    state.tasks.push({text: value, done: false});
    $("taskInput").value = "";
    save();
    renderTasks();
  });

  // Калькуляторы
  $("calcPercent").addEventListener("click", () => {
    const a = $("percentNumber").value.trim();
    const b = $("percentValue").value.trim();
    if (a === "" || b === "") {
      $("percentResult").textContent = "Введи число и процент.";
      return;
    }
    const n = Number(a), p = Number(b);
    if (!Number.isFinite(n) || !Number.isFinite(p)) {
      $("percentResult").textContent = "Проверь введённые числа.";
      return;
    }
    $("percentResult").textContent = `${p}% от ${n} = ${Number((n * p / 100).toFixed(6))}`;
  });

  $("calcAverage").addEventListener("click", () => {
    const raw = $("grades").value.trim();
    const result = $("averageResult");
    if (!raw) {
      result.textContent = "Введи оценки, например: 5, 4, 3, 5.";
      return;
    }
    const grades = raw.split(/[,\s;]+/).map(Number);
    if (grades.some(n => !Number.isFinite(n) || n < 1 || n > 5)) {
      result.textContent = "Введи оценки от 1 до 5 через запятую.";
      return;
    }
    result.textContent = "Твой средний балл: " +
      (grades.reduce((a, b) => a + b, 0) / grades.length).toFixed(2);
  });

  // Тест
  const questions = [
    {q:"Сколько будет 12 × 8?", options:["86","96","108"], answer:1},
    {q:"Сумма углов треугольника равна…", options:["90°","180°","360°"], answer:1},
    {q:"Сколько будет 15% от 200?", options:["15","20","30"], answer:2},
    {q:"Как найти скорость?", options:["v = s / t","v = m / V","F = m × a"], answer:0},
    {q:"Чему равен квадрат числа 7?", options:["14","49","77"], answer:1}
  ];

  function renderQuiz() {
    const root = $("quiz");
    root.replaceChildren();
    questions.forEach((q, i) => {
      const box = document.createElement("div");
      box.className = "formula";
      const title = document.createElement("strong");
      title.textContent = `${i + 1}. ${q.q}`;
      box.append(title);

      q.options.forEach((option, j) => {
        const label = document.createElement("label");
        const radio = document.createElement("input");
        radio.type = "radio";
        radio.name = "quiz-" + i;
        radio.value = j;
        radio.style.width = "auto";
        label.append(radio, document.createTextNode(" " + option));
        box.append(label);
      });
      root.append(box);
    });
  }

  $("checkQuiz").addEventListener("click", () => {
    if (state.quizDone) {
      $("quizResult").textContent = "Ты уже получил баллы за этот тест.";
      return;
    }
    let score = 0, answered = 0;
    questions.forEach((q, i) => {
      const selected = document.querySelector(`input[name="quiz-${i}"]:checked`);
      if (selected) {
        answered++;
        if (Number(selected.value) === q.answer) score++;
      }
    });
    if (answered < questions.length) {
      $("quizResult").textContent = "Ответь на все вопросы.";
      return;
    }
    $("quizResult").textContent = `Правильных ответов: ${score} из ${questions.length}.`;
    state.points += score * 3;
    state.quizDone = true;
    save();
  });

  // Формулы по классам
  const formulas = {
    4: [["Периметр прямоугольника","P = 2 × (a + b)"],["Площадь прямоугольника","S = a × b"],["Площадь квадрата","S = a × a"]],
    5: [["Процент","1% = 1/100 числа"],["Среднее арифметическое","Сумма чисел ÷ количество чисел"],["Периметр","Сумма длин всех сторон"]],
    6: [["Пропорция","a/b = c/d → a × d = b × c"],["Процент от числа","Число × процент ÷ 100"],["Скорость","v = s/t"]],
    7: [["Раскрытие скобок","a(b + c) = ab + ac"],["Квадрат суммы","(a+b)² = a²+2ab+b²"],["Линейное уравнение","ax + b = c → x = (c-b)/a, если a ≠ 0"]],
    8: [["Теорема Пифагора","c² = a² + b²"],["Разность квадратов","a² − b² = (a−b)(a+b)"],["Дискриминант","D = b² − 4ac"]],
    9: [["Квадратное уравнение","ax² + bx + c = 0"],["Корни квадратного уравнения","x = (−b ± √D)/(2a), если D ≥ 0 и a ≠ 0"],["Скорость","v = s/t"],["Закон Ома","I = U/R"]]
  };

  function renderFormulas() {
    const root = $("formulaList");
    root.replaceChildren();
    formulas[$("formulaGrade").value].forEach(([name, formula]) => {
      const div = document.createElement("div");
      div.className = "formula";
      const title = document.createElement("strong");
      title.textContent = name;
      const code = document.createElement("code");
      code.textContent = formula;
      div.append(title, code);

  function answerQuestion(question, grade) {
    const q = question.toLowerCase().replaceAll("ё", "е");

    // Простые арифметические примеры, например: 25 * 4
    const math = q.match(/(-?\d+(?:[.,]\d+)?)\s*([+*/×÷−-])\s*(-?\d+(?:[.,]\d+)?)/);
    if (math) {
      const a = Number(math[1].replace(",", "."));
      const b = Number(math[3].replace(",", "."));
      const op = math[2];
      let result;
      if (op === "+") result = a + b;
      else if (op === "-" || op === "−") result = a - b;
      else if (op === "*" || op === "×") result = a * b;
      else if (op === "/" || op === "÷") {
        if (b === 0) return "На ноль делить нельзя. Проверь пример.";
        result = a / b;
      }

      if (Number.isFinite(result)) {
        return `🧮 Разберём пример для ${grade} класса.\n\n${a} ${op} ${b} = ${Number(result.toFixed(8))}.\n\nПопробуй решить похожий пример самостоятельно, чтобы закрепить правило.`;
      }
    }

    for (const item of answers) {
      if (item.keys.some(key => q.includes(key))) {
        return `📚 Объяснение для ${grade} класса\n\n${item.text}\n\nПопробуй применить правило к своему примеру.`;
      }
    }

    if (q.includes("привет") || q.includes("здравств")) {
      return "Привет! 👋 Я учебный помощник SmartStudy. Спроси меня про дроби, проценты, уравнения, степени, площадь, периметр или формулы физики.";
    }

    return `Я пока работаю в демонстрационном режиме и не умею понимать любые вопросы как полноценная нейросеть.\n\nМогу помочь с такими темами:\n• Дроби и проценты\n• Простые арифметические примеры\n• Уравнения\n• Степени\n• Теорема Пифагора\n• Площадь и периметр\n• Основные формулы физики\n\nНапиши, например: «Объясни проценты» или «25 × 4».`;
  }
